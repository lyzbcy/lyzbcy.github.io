// 上饶攻略「密文附件」模式：整份自包含 HTML 加密存放于 /assets/encrypted/wuyuan-ppt.json，
// 解密在浏览器本地完成（PBKDF2 + AES-256-GCM），页面与 JS 中均无明文。
// 本脚本覆盖 encrypted-post.js 的默认 checkPassword。
(function () {
  'use strict';

  function hexToBytes(hex) {
    var bytes = new Uint8Array(hex.length / 2);
    for (var i = 0; i < bytes.length; i++) bytes[i] = parseInt(hex.substr(i * 2, 2), 16);
    return bytes;
  }
  function bytesToHex(bytes) {
    var s = '';
    for (var i = 0; i < bytes.length; i++) s += ('00' + bytes[i].toString(16)).slice(-2);
    return s;
  }

  var btnStyle = 'padding: 10px 22px; border-radius: 8px; border: none; background: #ff7bac; color: #fff; cursor: pointer; font-weight: bold; font-size: 14px; box-shadow: 0 4px 10px rgba(255,123,172,0.3); text-decoration: none; display: inline-block;';
  var blobUrl = null;

  window.checkPassword = async function () {
    var input = document.getElementById('partial-password-input');
    var errEl = document.getElementById('partial-error-message');
    var overlay = document.getElementById('partial-overlay');
    var target = document.getElementById('partial-protected-content');
    if (!input || !target) return;

    var pwd = input.value.trim();
    if (!pwd) { errEl.textContent = '请输入密码哦～ 💕'; input.focus(); return; }

    errEl.style.color = '#888';
    errEl.textContent = '🔓 正在本地解密攻略文件（约 2.6MB，稍等几秒）...';

    try {
      var res = await fetch('/assets/encrypted/wuyuan-ppt.json');
      if (!res.ok) throw new Error('密文文件加载失败（' + res.status + '）');
      var payload = await res.json();

      // PBKDF2 派生密钥，先比对 verify 再解密（与全站加密方案一致）
      var keyMaterial = await crypto.subtle.importKey(
        'raw', new TextEncoder().encode(pwd), 'PBKDF2', false, ['deriveKey', 'deriveBits']
      );
      var params = { name: 'PBKDF2', salt: hexToBytes(payload.salt), iterations: payload.iterations, hash: 'SHA-256' };
      var aesKey = await crypto.subtle.deriveKey(
        params, keyMaterial, { name: 'AES-GCM', length: 256 }, false, ['decrypt']
      );
      var verifyHex = bytesToHex(new Uint8Array(await crypto.subtle.deriveBits(params, keyMaterial, 256)));
      if (verifyHex !== payload.verify) {
        errEl.style.color = '';
        errEl.textContent = '密码不对哦，再想想～ 🤔';
        input.value = ''; input.focus();
        return;
      }

      var buf = await crypto.subtle.decrypt(
        { name: 'AES-GCM', iv: hexToBytes(payload.iv) }, aesKey, hexToBytes(payload.ciphertext)
      );
      var deckHtml = new TextDecoder('utf-8').decode(buf);
      blobUrl = URL.createObjectURL(new Blob([deckHtml], { type: 'text/html' }));

      // 渲染操作区
      overlay.style.display = 'none';
      target.style.display = 'block';
      document.getElementById('wuyuan-viewer').innerHTML =
        '<h3 style="color: #ff7bac; border-bottom: none; margin-bottom: 8px;">🎉 解锁成功！</h3>' +
        '<p style="color: #666; font-size: 14px; margin: 8px 0 16px;">整套 Web PPT 已加密保存在本博客里，与原服务器彻底解耦。</p>' +
        '<div style="display: flex; justify-content: center; gap: 10px; flex-wrap: wrap;">' +
        '<button id="wuyuanPreviewBtn" style="' + btnStyle + '">📖 本页预览</button>' +
        '<button id="wuyuanOpenBtn" style="' + btnStyle + '">🔗 新标签页打开</button>' +
        '<a id="wuyuanDlBtn" href="' + blobUrl + '" download="上饶五一攻略_自包含版.html" style="' + btnStyle + '">⬇️ 下载留存</a>' +
        '</div>' +
        '<p style="color: #999; font-size: 12px; margin-top: 12px;">💡 美食/照片日记保存在浏览器本地；地图导航为在线链接；建议电脑端查看。</p>';

      document.getElementById('wuyuanPreviewBtn').onclick = function () {
        document.getElementById('wuyuan-frame').src = blobUrl;
        document.getElementById('wuyuan-frame-wrap').style.display = 'block';
        this.textContent = '🔄 刷新预览';
      };
      document.getElementById('wuyuanOpenBtn').onclick = function () { window.open(blobUrl, '_blank'); };
    } catch (e) {
      errEl.style.color = '';
      errEl.textContent = '解密失败 😢 ' + (e && e.message ? e.message : '');
      input.value = ''; input.focus();
    }
  };
})();
