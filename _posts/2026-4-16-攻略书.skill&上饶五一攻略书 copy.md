    window.saveDiary = function(id) {
      const msg = document.getElementById('dmsg_' + id);
      try {
        let s = JSON.parse(localStorage.getItem('food_' + id) || '{}');

        msg.style.color = '#2b5e2b';
        renderFoodGrid();
      } catch(e) {
        msg.textContent = '❌ 存储已满';
        msg.style.color = 'red';
      }
      setTimeout(() => { msg.textContent = ''; }, 3000);
    };

    window.uploadMedia = function(input, type, id) {
      if (!input.files || !input.files[0]) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        let s = JSON.parse(localStorage.getItem('food_' + id) || '{}');
        s[type] = e.target.result;
        try {
          localStorage.setItem('food_' + id, JSON.stringify(s));
          openFoodModal(id);
          renderFoodGrid();
        } catch(err) {
          alert('图片/音频体积超出浏览器本地存储限制。部署后端后可无限上传！');
        }
      };
      reader.readAsDataURL(input.files[0]);
    };

    function showFoodHint(e, popupId) {
      e.stopPropagation();
      e.preventDefault();
      // Close all other food hint popups
      document.querySelectorAll('.food-hint-popup.show').forEach(p => p.classList.remove('show'));
      const popup = document.getElementById(popupId);
      // Move popup to body so it's not clipped by table/slide overflow
      if (popup.parentNode !== document.body) {
        document.body.appendChild(popup);
      }
      // Position near the button
      const btn = e.currentTarget;
      const rect = btn.getBoundingClientRect();
      popup.style.left = Math.max(10, rect.left - 60) + 'px';
      popup.style.top = Math.max(10, rect.top - popup.offsetHeight - 10) + 'px';
      popup.classList.add('show');
      // Re-adjust after it's visible (now offsetHeight is real)
      requestAnimationFrame(() => {
        popup.style.top = Math.max(10, rect.top - popup.offsetHeight - 10) + 'px';
      });
    }
    // Close food hint popup when clicking anywhere else
    document.addEventListener('click', function(e) {
      if (!e.target.closest('.food-hint') &amp;&amp; !e.target.closest('.food-hint-popup')) {
        document.querySelectorAll('.food-hint-popup.show').forEach(p => p.classList.remove('show'));
      }
    });

  function renderPhotoGrid() {
    const container = document.getElementById('photo-grid-container');
    if (!container) return;
    container.innerHTML = '';
    
    // Group photos by location
    const groups = {};
    PHOTOS.forEach(photo => {
      if (!groups[photo.where]) groups[photo.where] = [];
      groups[photo.where].push(photo);
    });

    let isFirst = true;
    for (const [location, photos] of Object.entries(groups)) {
      const section = document.createElement('div');
      section.style.marginBottom = '15px';
      section.style.background = '#fafafa';
      section.style.borderRadius = '12px';
      section.style.overflow = 'hidden';
      section.style.border = '1px solid #eaeaea';
      
      const header = document.createElement('div');
      header.style.padding = '12px 16px';
      header.style.background = '#f0e6f6';
      header.style.cursor = 'pointer';
      header.style.display = 'flex';
      header.style.justifyContent = 'space-between';
      header.style.alignItems = 'center';
      header.style.fontWeight = 'bold';
      header.style.color = '#8e44ad';
      header.style.userSelect = 'none';
      
      const grid = document.createElement('div');
      grid.className = 'food-grid';
      grid.style.padding = '15px';
      grid.style.marginTop = '0'; // Override any margin
      // Let the first folder be expanded, others collapsed
      grid.style.display = isFirst ? 'grid' : 'none';

      header.innerHTML = `<span>📁 ${location} <span style=&quot;font-size:0.8em; color:#999; font-weight:normal; margin-left:6px;&quot;>(${photos.length}个机位)</span></span><span class=&quot;folder-icon&quot;>${isFirst ? '▲' : '▼'}</span>`;
      
      header.onclick = () => {
        const isHidden = grid.style.display === 'none';
        grid.style.display = isHidden ? 'grid' : 'none';
        header.querySelector('.folder-icon').textContent = isHidden ? '▲' : '▼';
        // Optional: change header background dynamically when active
        header.style.background = isHidden ? '#e9d5f3' : '#f0e6f6';
      };
      if (isFirst) header.style.background = '#e9d5f3'; // active color for the first one

      photos.forEach(photo => {
        const saved = JSON.parse(localStorage.getItem('photo_' + photo.id) || '{}');
        const tile = document.createElement('div');
        tile.className = `food-tile${saved.shot ? ' eaten' : ''}`;
        tile.onclick = () => openPhotoModal(photo.id);
        tile.innerHTML = `
          <img class=&quot;food-tile-img&quot; src=&quot;${saved.img || photo.img}&quot; alt=&quot;${photo.name}&quot;>
          <div class=&quot;food-tile-body&quot;>
            <div class=&quot;food-tile-name&quot;>${photo.emoji} ${photo.name}</div>
            <div class=&quot;food-tile-tag&quot;>${photo.tag}</div>
            ${saved.shot ? '<div style=&quot;color:#2b5e2b; font-size:0.8em; font-weight:bold; margin-top:4px;&quot;>✅ 已拍到！</div>' : ''}
          </div>
        `;
        grid.appendChild(tile);
      });
      
      section.appendChild(header);
      section.appendChild(grid);
      container.appendChild(section);
      
      isFirst = false;
    }
  }

  window.toggleShot = function(id, checked) {
    let s = JSON.parse(localStorage.getItem('photo_' + id) || '{}');
    s.shot = checked;
    localStorage.setItem('photo_' + id, JSON.stringify(s));
    renderPhotoGrid();
    openPhotoModal(id);
  };
  window.savePhotoDiary = function(id) {
    const msg = document.getElementById('pmsg_' + id);
    try {
      let s = JSON.parse(localStorage.getItem('photo_' + id) || '{}');
      s.review = document.getElementById('photodiary_' + id).value;
      localStorage.setItem('photo_' + id, JSON.stringify(s));
      msg.textContent = '✅ 已保存！'; msg.style.color = '#8e44ad';
      renderPhotoGrid();
    } catch(e) { msg.textContent = '❌ 存储已满'; msg.style.color = 'red'; }
    setTimeout(() => { msg.textContent = ''; }, 3000);
  };
  window.uploadPhotoMedia = function(input, type, id) {
    if (!input.files || !input.files[0]) return;
    const reader = new FileReader();
    reader.onload = e => {
      let s = JSON.parse(localStorage.getItem('photo_' + id) || '{}');
      s[type] = e.target.result;
      try { localStorage.setItem('photo_' + id, JSON.stringify(s)); openPhotoModal(id); renderPhotoGrid(); }
      catch(err) { alert('图片体积超出限制，部署后端后可无限上传！'); }
    };
    reader.readAsDataURL(input.files[0]);
  };
  window.deletePhotoMedia = function(id, type) {
    let s = JSON.parse(localStorage.getItem('photo_' + id) || '{}');
    delete s[type]; localStorage.setItem('photo_' + id, JSON.stringify(s));
    openPhotoModal(id); renderPhotoGrid();
  };
  window.clearPhotoLog = function(id) {
    if (!confirm('确定清空这个机位的所有日记吗？')) return;
    localStorage.removeItem('photo_' + id);
    openPhotoModal(id); renderPhotoGrid();
  };
