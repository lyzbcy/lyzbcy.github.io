document.addEventListener('DOMContentLoaded', function() {
  const modal = document.getElementById('drinkModal');
  const modalTitle = document.getElementById('modalTitle');
  const modalBody = document.getElementById('modalBody');
  const modalClose = document.getElementById('modalClose');
  
  // 品类模态框元素
  const categoryModal = document.getElementById('categoryModal');
  const categoryIcon = document.getElementById('categoryIcon');
  const categoryTitle = document.getElementById('categoryTitle');
  const categoryDesc = document.getElementById('categoryDesc');
  const categoryBody = document.getElementById('categoryBody');
  const categoryClose = document.getElementById('categoryClose');
  
  // 从卡片内容中解析品类
  function extractCategory(cardText) {
    const lines = cardText.split('\n').map(l => l.replace(/<br>/gi, '\n')).join('\n').split('\n');
    // 常见品类关键词
    const categories = ['能量饮料', '电解质水', '运动饮料'];
    for (const line of lines) {
      for (const cat of categories) {
        if (line.includes(cat)) return cat;
      }
    }
    return null;
  }
  
  // 获取某品类下的所有产品，按评级排序
  function getProductsByCategory(category) {
    const products = [];
    for (const [key, value] of Object.entries(drinkData)) {
      if (value.category === category) {
        products.push({ key, ...value });
      }
    }
    // 按评级排序
    products.sort((a, b) => (tierOrder[a.tier] || 5) - (tierOrder[b.tier] || 5));
    return products;
  }
  
  // 生成品类产品列表HTML
  function generateCategoryContent(products) {
    if (products.length === 0) {
      return `
        <div class="empty-category">
          <div class="empty-category-icon">📭</div>
          <p>暂无该品类的产品评测</p>
        </div>
      `;
    }
    
    return products.map((product, index) => `
      <div class="product-item" data-index="${index}">
        <div class="product-header">
          <span class="product-tier-badge ${tierClass[product.tier] || 'tier-average'}">
            ${tierEmoji[product.tier] || '⚪'} ${product.tier}
          </span>
          <span class="product-name">${product.title}</span>
          <span class="product-expand-icon">▼</span>
        </div>
        <div class="product-content">
          ${product.content}
        </div>
      </div>
    `).join('');
  }

  document.querySelectorAll('.drink-card').forEach(card => {
    card.addEventListener('click', function(e) {
      const drinkKey = this.getAttribute('data-drink');
      const data = drinkData[drinkKey];
      
      if (data && data.category) {
        // 显示品类模态框
        const catData = categoryData[data.category] || { icon: '📦', description: '' };
        const products = getProductsByCategory(data.category);
        
        categoryIcon.textContent = catData.icon;
        categoryTitle.textContent = data.category;
        categoryDesc.textContent = catData.description;
        categoryBody.innerHTML = generateCategoryContent(products);
        
        // 为产品项添加展开/收起事件
        categoryBody.querySelectorAll('.product-item').forEach(item => {
          item.querySelector('.product-header').addEventListener('click', function() {
            item.classList.toggle('expanded');
          });
        });
        
        // 自动展开当前点击的产品
        const currentIndex = products.findIndex(p => p.key === drinkKey);
        if (currentIndex >= 0) {
          const currentItem = categoryBody.querySelector(`.product-item[data-index="${currentIndex}"]`);
          if (currentItem) {
            currentItem.classList.add('expanded');
            // 滚动到当前项
            setTimeout(() => {
              currentItem.scrollIntoView({ behavior: 'smooth', block: 'center' });
            }, 100);
          }
        }
        
        categoryModal.classList.add('show');
        document.body.style.overflow = 'hidden';
      } else if (data) {
        // 无品类信息时显示原有模态框
        modalTitle.textContent = data.title;
        modalBody.innerHTML = data.content;
        modal.classList.add('show');
        document.body.style.overflow = 'hidden';
      }
    });
  });
  
  // 关闭饮料模态框
  modalClose.addEventListener('click', function() {
    modal.classList.remove('show');
    document.body.style.overflow = '';

  // 关闭品类模态框

    categoryModal.classList.remove('show');
    document.body.style.overflow = '';
  });

    if (e.target === modal) {

      document.body.style.overflow = '';
    }
  });

      categoryModal.classList.remove('show');
      document.body.style.overflow = '';
    }
  });
  
  // ESC 键关闭模态框

    if (e.key === 'Escape') {

        document.body.style.overflow = '';
      }
      if (categoryModal.classList.contains('show')) {
        categoryModal.classList.remove('show');
        document.body.style.overflow = '';
      }
    }
  });
